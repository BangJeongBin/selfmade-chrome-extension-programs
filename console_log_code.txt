const targets = ['데이터1', '데이터2', '데이터3'];

const results = [];

document.querySelectorAll('ul.reply-data').forEach(el => {
    const items = el.querySelectorAll('li');

    const writer = items[0]?.querySelector('.txt')?.innerText.trim() || '';

    if (targets.some(target => writer.includes(target))) {

        const date = items[1]?.querySelector('.txt')?.innerText.trim() || '';
        const comment = el.querySelector('.comment.show')?.innerText.trim() || '';

        results.push({
            작성자: writer,
            작성일시: date,
            댓글: comment
        });
    }
});

console.table(results);
